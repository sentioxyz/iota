        // The issuer of the <Link to="../stardust/nft#stardust_nft_NFT">NFT</Link>. Equivalent to IRC-27 `collectionId`.
        string::utf8(b"\{<Link to="../stardust/nft#stardust_nft_immutable_issuer">immutable_issuer</Link>}"),
    ];
    <b>let</b> <b>mut</b> display = display::new_with_fields&lt;<Link to="../stardust/nft#stardust_nft_Nft">Nft</Link>&gt;(

        keys,
        values,
        ctx,

    // Commit the first version of `Display` to apply changes.
    display.update_version();

    package::burn_publisher(publisher);

}
