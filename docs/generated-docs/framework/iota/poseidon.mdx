## Function `poseidon_bn254`

@param data: Vector of BN254 field elements to hash.

Hash the inputs using poseidon_bn254 and returns a BN254 field element.

Each element has to be a BN254 field element in canonical representation so it must be smaller than the BN254
scalar field size which is 21888242871839275222246405745257275088548364400416034343698204186575808495617.

<b>public</b> <b>fun</b> <Link to="../iota/poseidon#iota_poseidon_poseidon_bn254">poseidon_bn254</Link>(data: &vector&lt;u256&gt;): u256 \{
    <b>let</b> (<b>mut</b> i, <b>mut</b> b, l) = (0, vector[], data.length());
    <b>assert</b>!(l &gt; 0, <Link to="../iota/poseidon#iota_poseidon_EEmptyInput">EEmptyInput</Link>);
    <b>while</b> (i &lt; l) \{
        <b>let</b> field_element = &data[i];
        <b>assert</b>!(*field_element &lt; <Link to="../iota/poseidon#iota_poseidon_BN254_MAX">BN254_MAX</Link>, <Link to="../iota/poseidon#iota_poseidon_ENonCanonicalInput">ENonCanonicalInput</Link>);
        b.push_back(<Link to="../iota/bcs#iota_bcs_to_bytes">bcs::to_bytes</Link>(&data[i]));
        i = i + 1;
    };
    <b>let</b> binary_output = <Link to="../iota/poseidon#iota_poseidon_poseidon_bn254_internal">poseidon_bn254_internal</Link>(&b);
    <Link to="../iota/bcs#iota_bcs_new">bcs::new</Link>(binary_output).peel_u256()
}
