## Struct `ValidatorSetV2`

The second version of the struct storing information about validator set.
This version is an extension on the first one, that supports a new approach to committee selection,
where committee members taking part in consensus are selected from a set of <code><Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link></code>
before an epoch begins. <code>committee_members</code> is a vector of indices of validators stored in <code><Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link></code>,
that have been selected to take part in consensus during the current epoch.

    validator_report_records: &<b>mut</b> VecMap&lt;<b>address</b>, VecSet&lt;<b>address</b>&gt;&gt;,
    committee_addresses: vector&lt;<b>address</b>&gt;,
    ctx: &<b>mut</b> TxContext,
) \{
    // Iterate through all the active validators, record their low stake status, and kick them out <b>if</b> the condition is met.
    <b>let</b> <b>mut</b> i = self.<Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link>.length();
    <b>while</b> (i &gt; 0) \{
        i = i - 1;
        <b>let</b> validator_ref = &self.<Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link>[i];
        <b>let</b> validator_address = validator_ref.iota_address();
        <b>let</b> stake = validator_ref.total_stake_amount();
        <b>if</b> (stake &gt;= low_stake_threshold) \{
            // The <Link to="../iota_system/validator#iota_system_validator">validator</Link> is safe. We remove their <b>entry</b> from the at_risk map <b>if</b> there exists one.
            <b>if</b> (self.at_risk_validators.contains(&validator_address)) \{
                self.at_risk_validators.remove(&validator_address);
            }
        } <b>else</b> <b>if</b> (stake &gt;= very_low_stake_threshold) \{
            // The stake is a bit below the threshold so we increment the <b>entry</b> of the <Link to="../iota_system/validator#iota_system_validator">validator</Link> in the map.
            <b>let</b> new_low_stake_period = <b>if</b> (self.at_risk_validators.contains(&validator_address)) \{
                <b>let</b> num_epochs = &<b>mut</b> self.at_risk_validators[&validator_address];
                *num_epochs = *num_epochs + 1;
                *num_epochs
            } <b>else</b> \{
                self.at_risk_validators.insert(validator_address, 1);
                1
            };
            // If the grace period <b>has</b> passed, the <Link to="../iota_system/validator#iota_system_validator">validator</Link> <b>has</b> to leave us.
            <b>if</b> (new_low_stake_period &gt; low_stake_grace_period) \{
                <b>let</b> <Link to="../iota_system/validator#iota_system_validator">validator</Link> = self.<Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link>.remove(i);
                <b>let</b> is_committee = committee_addresses.contains(&<Link to="../iota_system/validator#iota_system_validator">validator</Link>.iota_address());
                <Link to="../iota_system/validator_set#iota_system_validator_set_process_validator_departure">process_validator_departure</Link>(
                    self,
                    <Link to="../iota_system/validator#iota_system_validator">validator</Link>,
                    validator_report_records,
                    <b>false</b>,
                    /* the <Link to="../iota_system/validator#iota_system_validator">validator</Link> is kicked out involuntarily */
                    is_committee,
                    ctx,
                );
            }
        } <b>else</b> \{
            // The <Link to="../iota_system/validator#iota_system_validator">validator</Link>'s stake is lower than the very low threshold so we kick them out immediately.
            <b>let</b> <Link to="../iota_system/validator#iota_system_validator">validator</Link> = self.<Link to="../iota_system/validator_set#iota_system_validator_set_active_validators">active_validators</Link>.remove(i);
            <b>let</b> is_committee = committee_addresses.contains(&<Link to="../iota_system/validator#iota_system_validator">validator</Link>.iota_address());
            <Link to="../iota_system/validator_set#iota_system_validator_set_process_validator_departure">process_validator_departure</Link>(
                self,
                <Link to="../iota_system/validator#iota_system_validator">validator</Link>,
                validator_report_records,
                <b>false</b>,
                /* the <Link to="../iota_system/validator#iota_system_validator">validator</Link> is kicked out involuntarily */
                is_committee,
                ctx,
            );
        }
    }
}

## Function `derive_reference_gas_price`

The returned gas price should be greater than or equal to 2/3 of the validators submitted

    // Build a priority queue that will pop entries with gas price from the highest to the lowest.

    <b>let</b> <b>mut</b> result = 0;

        <b>let</b> (gas_price, <Link to="../iota_system/voting_power#iota_system_voting_power">voting_power</Link>) = pq.pop_max();

<b>fun</b> <Link to="../iota_system/validator_set#iota_system_validator_set_compute_adjusted_reward_distribution">compute_adjusted_reward_distribution</Link>(
    committee_members: &vector&lt;u64&gt;,
    unadjusted_staking_reward_amounts: vector&lt;u64&gt;,
    slashed_validator_indices_set: VecSet&lt;u64&gt;,
    reward_slashing_rate: u64,
): vector&lt;u64&gt; \{
    <b>let</b> <b>mut</b> adjusted_staking_reward_amounts = vector[];
    // Loop through each <Link to="../iota_system/validator#iota_system_validator">validator</Link> and adjust rewards <b>as</b> necessary
    <b>let</b> length = committee_members.length();
    <b>let</b> <b>mut</b> i = 0;
    <b>while</b> (i &lt; length) \{
        <b>let</b> unadjusted_staking_reward_amount = unadjusted_staking_reward_amounts[i];
        // Check <b>if</b> the <Link to="../iota_system/validator#iota_system_validator">validator</Link> is slashed
        <b>let</b> adjusted_staking_reward_amount = <b>if</b> (
            slashed_validator_indices_set.contains(&committee_members[i])
        ) \{
            // Use the slashing rate to compute the amount of staking rewards slashed from this punished <Link to="../iota_system/validator#iota_system_validator">validator</Link>.
            // Use u128 to avoid multiplication overflow.
            <b>let</b> staking_reward_adjustment_u128 =
                ((unadjusted_staking_reward_amount <b>as</b> u128) * (reward_slashing_rate <b>as</b> u128)) / <Link to="../iota_system/validator_set#iota_system_validator_set_BASIS_POINT_DENOMINATOR">BASIS_POINT_DENOMINATOR</Link>;
            unadjusted_staking_reward_amount - (staking_reward_adjustment_u128 <b>as</b> u64)
        } <b>else</b> \{
            // Otherwise, unadjusted staking reward amount is assigned to the unslashed validators
            unadjusted_staking_reward_amount
        };
        adjusted_staking_reward_amounts.push_back(adjusted_staking_reward_amount);
        // Move to the next <Link to="../iota_system/validator#iota_system_validator">validator</Link>
        i = i + 1;
    };
    // The sum of the adjusted staking rewards may not be equal to the total staking reward,
    // because of integer division truncation and the slashing of the rewards <b>for</b> the slashed validators.
    adjusted_staking_reward_amounts
