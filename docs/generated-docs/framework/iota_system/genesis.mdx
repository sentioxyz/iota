## Function `create`

This function will be explicitly called once at genesis.
It will create a singleton IotaSystemState object, which contains
all the information we need in the system.

<pre>
<code>
<b>fun</b> <Link to="../iota_system/genesis#iota_system_genesis_create">create</Link>(
    iota_system_state_id: UID,
    <b>mut</b> iota_treasury_cap: IotaTreasuryCap,
    genesis_chain_parameters: <Link to="../iota_system/genesis#iota_system_genesis_GenesisChainParameters">GenesisChainParameters</Link>,
    genesis_validators: vector&lt;<Link to="../iota_system/genesis#iota_system_genesis_GenesisValidatorMetadata">GenesisValidatorMetadata</Link>&gt;,
    token_distribution_schedule: <Link to="../iota_system/genesis#iota_system_genesis_TokenDistributionSchedule">TokenDistributionSchedule</Link>,
    timelock_genesis_label: Option&lt;String&gt;,
    iota_system_admin_cap: IotaSystemAdminCap,
    ctx: &<b>mut</b> TxContext,
) \{
    // Ensure this is only called at <Link to="../iota_system/genesis#iota_system_genesis">genesis</Link>
    <b>assert</b>!(ctx.epoch() == 0, <Link to="../iota_system/genesis#iota_system_genesis_ENotCalledAtGenesis">ENotCalledAtGenesis</Link>);
    <b>let</b> <Link to="../iota_system/genesis#iota_system_genesis_TokenDistributionSchedule">TokenDistributionSchedule</Link> \{
        pre_minted_supply,
        allocations,
    } = token_distribution_schedule;
    <b>assert</b>!(iota_treasury_cap.total_supply() == pre_minted_supply, <Link to="../iota_system/genesis#iota_system_genesis_EWrongPreMintedSupply">EWrongPreMintedSupply</Link>);
    <b>let</b> <Link to="../iota_system/storage_fund#iota_system_storage_fund">storage_fund</Link> = balance::zero();
    // Create all the `ValidatorV1` structs
    <b>let</b> <b>mut</b> validators = vector[];
    <b>let</b> count = genesis_validators.length();
    <b>let</b> <b>mut</b> i = 0;
    <b>while</b> (i &lt; count) \{
        <b>let</b> <Link to="../iota_system/genesis#iota_system_genesis_GenesisValidatorMetadata">GenesisValidatorMetadata</Link> \{
            name,
            description,
            image_url,
            project_url,
            iota_address,
            gas_price,
            commission_rate,
            authority_public_key,
            proof_of_possession,
            network_public_key,
            protocol_public_key,
            network_address,
            p2p_address,
            primary_address,
        } = genesis_validators[i];
        <b>let</b> <Link to="../iota_system/validator#iota_system_validator">validator</Link> = <Link to="../iota_system/validator#iota_system_validator_new">validator::new</Link>(
            iota_address,
            authority_public_key,
            network_public_key,
            protocol_public_key,
            proof_of_possession,
            name,
            description,
            image_url,
            project_url,
            network_address,
            p2p_address,
            primary_address,
            gas_price,
            commission_rate,
            ctx,
        );
        // Ensure that each <Link to="../iota_system/validator#iota_system_validator">validator</Link> is unique
        <b>assert</b>!(
            !<Link to="../iota_system/validator_set#iota_system_validator_set_is_duplicate_validator">validator_set::is_duplicate_validator</Link>(&validators, &<Link to="../iota_system/validator#iota_system_validator">validator</Link>),
            <Link to="../iota_system/genesis#iota_system_genesis_EDuplicateValidator">EDuplicateValidator</Link>,
        );
        validators.push_back(<Link to="../iota_system/validator#iota_system_validator">validator</Link>);
        i = i + 1;
    };
    // Allocate tokens and staking operations
    <Link to="../iota_system/genesis#iota_system_genesis_allocate_tokens">allocate_tokens</Link>(
        &<b>mut</b> iota_treasury_cap,
        allocations,
        &<b>mut</b> validators,
        timelock_genesis_label,
        ctx,
    );
    // Activate all validators
    <Link to="../iota_system/genesis#iota_system_genesis_activate_validators">activate_validators</Link>(&<b>mut</b> validators);
    <b>let</b> system_parameters = <Link to="../iota_system/iota_system_state_inner#iota_system_iota_system_state_inner_create_system_parameters">iota_system_state_inner::create_system_parameters</Link>(
        genesis_chain_parameters.epoch_duration_ms,
        // ValidatorV1 committee parameters
        genesis_chain_parameters.max_validator_count,
        genesis_chain_parameters.min_validator_joining_stake,
        genesis_chain_parameters.validator_low_stake_threshold,
        genesis_chain_parameters.validator_very_low_stake_threshold,
        genesis_chain_parameters.validator_low_stake_grace_period,
        ctx,
    );
    iota_system::create(
        iota_system_state_id,
        iota_treasury_cap,
        validators,
        <Link to="../iota_system/storage_fund#iota_system_storage_fund">storage_fund</Link>,
        genesis_chain_parameters.protocol_version,
        genesis_chain_parameters.chain_start_timestamp_ms,
        system_parameters,
        iota_system_admin_cap,
        ctx,
    );
}

                <Link to="../iota_system/validator#iota_system_validator">validator</Link>.request_add_stake_at_genesis(

                    ctx,
                );

            <Link to="../iota/transfer#iota_transfer">iota::transfer</Link>(
                allocation_balance.into_coin(ctx),

        };
    }; allocations.destroy_empty(); }
